bool isFunction(const std::string& token) {
    static const std::unordered_set<std::string> functions = {"sin", "cos", "sqrt", "pow"};
    return functions.find(token) != functions.end();
}
// 修改 precedence 函数，增加函数优先级（可用特殊值）
int precedence(const std::string& op) {
    if (op == "sin" || op == "cos" || op == "sqrt") return 4;  // 单目函数
    if (op == "pow") return 3;  // 双目函数，优先级低于单目但高于乘除
    if (op == "~") return 4;
    if (op == "+" || op == "-") return 1;
    if (op == "*" || op == "/") return 2;
    return 0;
}

Q3.	输入sin(Π/3)程序报错运行中止
A3.	到了非 ASCII 字符 Π（其 ASCII 为 -90），导致 isalpha(c) 判断出错，并且无法处理该字符。
修改 ShuntingYard.cpp 中的字符判断：将 isalpha(c) 改为显式检查 ASCII 字母范围，避免负值字符被误判：
bool isAsciiAlpha(char c) {
    return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z');
}
然后在循环中将 else if (isalpha(c)) 替换为 else if (isAsciiAlpha(c))。
支持数学常量 pi 和 e：识别标识符后，检查是否为常量，如果是则直接当作数字处理（推入输出队列）。
添加常量判断函数：
bool isConstant(const std::string& token) {
    return token == "pi" || token == "e";
}

double getConstantValue(const std::string& token) {
    if (token == "pi") return 3.141592653589793;
    if (token == "e") return 2.718281828459045;
    return 0.0;
}
修改标识符处理逻辑：
else if (isAsciiAlpha(c)) {

    std::string ident;
